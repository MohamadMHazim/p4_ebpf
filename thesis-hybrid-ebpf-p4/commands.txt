import struct

def crc16_ibm_reflected(data: bytes):
    crc = 0xFFFF
    for b in data:
        crc ^= b
        for _ in range(8):
            if crc & 1:
                crc = (crc >> 1) ^ 0xA001
            else:
                crc >>= 1
    return crc & 0xFFFF

def ip_to_bytes(ip):
    return bytes(map(int, ip.split(".")))

src_ip = "10.0.0.1"
dst_ip = "10.0.0.2"
src_port = 12345
dst_port = 443
protocol = 6  # TCP

data = b""
data += ip_to_bytes(src_ip)
data += ip_to_bytes(dst_ip)
data += struct.pack("!H", src_port)
data += struct.pack("!H", dst_port)
data += struct.pack("!B", protocol)

index = crc16_ibm_reflected(data)
print("flow_index =", index)

from scapy.all import Ether, IP, TCP, sendp

iface = "veth0"   # change if needed
pkt = Ether()/IP(src="10.0.0.1", dst="10.0.0.2")/TCP(sport=12345, dport=443, flags="S")
sendp(pkt, iface=iface, count=20, inter=0.05, verbose=False)
